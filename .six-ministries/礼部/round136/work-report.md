```typescript
// Widget类型定义
interface DashboardWidget {
  id: string;
  type: WidgetType;
  title: string;
  size: 'small' | 'medium' | 'large';
  position: { x: number; y: number };
  config: Record<string, any>;
  refreshInterval?: number;
}

type WidgetType = 
  | 'system-status'     // 系统状态
  | 'storage-pool'      // 存储池
  | 'nvme-health'       // NVMe健康
  | 'network-io'        // 网络IO
  | 'container-list'    // 容器列表
  | 'backup-status'     // 备份状态
  | 'ai-tasks'          // AI任务
  | 'alerts'            // 告警
  | 'quick-actions'     // 快捷操作
  | 'custom';           // 自定义
```
